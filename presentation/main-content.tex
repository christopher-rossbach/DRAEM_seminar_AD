% Introduction
\begin{frame}{Introduction}{Surface Anomaly Detection}
    \begin{itemize}[<+->]
        \item Detection and localization of anomalies in the image space
        \item E.g. RGB-drone images, medical scans, \ldots
        \item OOD-Samples are often rare and diverse
        \item Collecting data for all possible anomalies is often infeasible
        \item Data collection gets even harder when pixel-accurate localization is required
        \item Therefore, methods that only rely on normal data are preferred
    \end{itemize}
\end{frame}

\begin{frame}{Problem Statement}{Current Situation and Related Work}
        \begin{columns}
            \begin{column}{0.45\textwidth}
                \begin{itemize}
                    \item<1-> Two groups of methods in this area:
                \end{itemize}
                \begin{enumerate}
                    \item<2-> \textbf{Reconstructive Methods} (Autoencoders, GANs) detect anomalies by measuring reconstruction error
                    \begin{itemize}
                        \item<3-> Minor anomalies often well reconstructed
                    \end{itemize}
                    \item<4-> \textbf{Discriminative Methods} on (deep) features (e.g. CNNs) distribution deviation
                    \begin{itemize}
                        \item<5-> Often suffer from unsharp segmentation
                        \item<5-> Overfit to synthetic anomalies
                    \end{itemize}
                \end{enumerate}
            \end{column}

    \begin{columns}
        \begin{column}{0.5\textwidth}

                    \item Distinguish between crack, corrosion or just paint
